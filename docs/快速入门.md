# 快速入门

本文按模块组给出典型解决方案。每个模块组按主题拆成多个代码块，组内相邻片段可以组合，跨模块组不要拼成一个 `.c` 文件；片段只保留当前场景需要的头文件与逻辑。接口细节见对应模块文档，构建与自动初始化见 [开发手册](开发手册.md)。

> 库由 `crystal_std` 在 `main` 之前自动初始化、在 `main` 之后自动逆初始化；应用直接写业务代码，`init_*` / `deinit_*` 由库完成。Windows 链接包含网络模块的程序时使用 `-lws2_32 -liphlpapi`，构建脚本已经为测试与演示程序处理。

## 1. 基础类型组

### 1.1 数字、CRC 与通配符

```C
#include <cc_atoi.h>
#include <cc_crc.h>
#include <cc_wildcard.h>

static void parse_and_check(void)
{
    CCINT32 n = cc_atoi("-42");                 /* 格式非法返回 0；溢出返回 INT32_MAX */
    CCUINT32 crc = cc_crc32("hello", 5, 0);     /* 分块续算：cc_crc32(next, len, crc) */
    CCBOOL image = cc_wildcard("*.png", "logo.png");

    (void)n;
    (void)crc;
    (void)image;
}
```

要点：`cc_atoi` 用返回值区分“格式非法”和“溢出”的方式有限，业务输入需要更严格的校验时，先做字符串格式检查。

### 1.2 时间与睡眠

```C
#include <cctime.h>

static void time_once(void)
{
    double start = cctime();
    ccsleep(10000);                 /* 10 ms，单位微秒 */
    double cost = cctime() - start;
    CCDate now = ccdate();

    (void)cost;
    (void)now;
}
```

`cctime()` 返回秒级浮点时间戳；`ccdate()` 返回拆分的年月日时分秒与毫秒。

## 2. 彩色输出

```C
#include <color_print.h>

static void print_status(void)
{
    color_print(CC_TEXT_COLOR(CC_GREEN, CC_DEFAULT) "OK\n");

```

`CC_TEXT_COLOR(前景, 背景)` 生成 `%c_XY` 控制符；每次 `color_print` 结束会自动恢复终端默认颜色。

## 3. 分配器

```C
#include <ccalloc.h>

static void *grow_buffer(void)
{
    void *buf = ccalloc(NULL, 4096, "atlas");
    void *next;
    void *shrunk;

    if (!buf)
        return NULL;

    next = ccalloc(buf, 8192, "atlas v2");
    if (!next)
    {
        ccalloc(buf, 0, NULL);
        return NULL;
    }

    shrunk = ccalloc(next, 2048, NULL);   /* 收缩，保留注释 */
    if (!shrunk)
    {
        ccalloc(next, 0, NULL);
        return NULL;
    }
    return shrunk;
}

static void free_buffer(void *buf)
{
    ccalloc(buf, 0, NULL);
}
```

`ccalloc(old, size, comment)` 同时承担分配、扩容 / 收缩和释放：`old == NULL` 分配，`size == 0` 释放。注释只在分配器内部使用，进程退出时由 `deinit_ccalloc` 输出残留。

## 4. 动态数组

```C
#include <cc_dyn.h>
#include <ccalloc.h>

typedef struct point {
    CCINT32 x;
    CCINT32 y;
} Point;

static void points_once(void)
{
    CCDyn points = {0};
    Point in = {1, 2};
    Point out;

    if (cc_dyn_core_push(&points, sizeof(Point), sizeof(Point), &in) != 0)
        goto done;

    if (cc_dyn_core_get(&points, sizeof(Point), sizeof(Point), &out, 0) != 0)
        goto done;

    (void)cc_dyn_core_pop(&points, sizeof(Point), sizeof(Point), &out);

done:
    if (points.array)
        ccalloc(points.array, 0, NULL);   /* 数组内存由 ccalloc 管理 */
}
```

第二个参数 `align` 是**步长**：`align * index` 定位到字节偏移，`size` 是这个元素占用的字节数。元素大小固定时通常两者都填 `sizeof(T)`。

## 5. 红黑树

`cc_rbt_core` 不关心节点长什么样，只通过 `CC_RbtOps` 访问键、左右子节点和颜色。下面是一个最小可用的“字符串索引”适配器。

```C
#include <cc_rbt.h>
#include <ccalloc.h>
#include <stdio.h>
#include <string.h>

typedef struct record {
    CCUINTPTR key;
    CCBOOL red;
    struct record *left;
    struct record *right;
    struct record *parent;
    char name[32];
} Record;

typedef struct index {
    CC_RbtOps ops;
    Record *root;
} Index;
```

比较与数据拷贝：`copy_node` 只复制键和业务数据，结构指针由内核维护。

```C
static CCINT32 rec_cmp_key(void *user, void *node, CCUINTPTR key)
{
    Record *r = (Record *)node;
    (void)user;
    if (r->key == key)
        return 0;
    return r->key > key ? 1 : -1;
}

static CCINT32 rec_cmp_node(void *user, void *node_1, void *node_2)
{
    Record *a = (Record *)node_1;
    Record *b = (Record *)node_2;
    (void)user;
    if (a->key == b->key)
        return 0;
    return a->key > b->key ? 1 : -1;
}

static CCINT32 rec_copy_node(void *user, void *dst_node, void *src_node)
{
    Record *dst = (Record *)dst_node;
    Record *src = (Record *)src_node;
    (void)user;
    dst->key = src->key;
    memcpy(dst->name, src->name, sizeof(dst->name));
    return 0;
}

static void *rec_find_instead(void *user, void *node)
{
    Record *r = ((Record *)node)->right;
    (void)user;
    if (!r)
        return NULL;
    while (r->left)
        r = r->left;
    return r;
}
```

结构访问、颜色和旋转：

```C
static void *rec_get_node(void *user, void *node, CC_RbtRel relation)
{
    Record *r = (Record *)node;
    (void)user;
    switch (relation)
    {
    case CC_RBT_REL_LEFT:   return r->left;
    case CC_RBT_REL_RIGHT:  return r->right;
    case CC_RBT_REL_PARENT: return r->parent;
    default:                return NULL;
    }
}

static CCUINT32 rec_set_node(void *user, void *node_1, void *node_2,
                             CC_RbtRel relation)
{
    Record *a = (Record *)node_1;
    Record *b = (Record *)node_2;
    (void)user;
    switch (relation)
    {
    case CC_RBT_REL_LEFT:   a->left = b;   return 0;
    case CC_RBT_REL_RIGHT:  a->right = b;  return 0;
    case CC_RBT_REL_PARENT: a->parent = b; return 0;
    default:                return 1;
    }
}

static CCBOOL rec_get_color(void *user, void *node)
{
    (void)user;
    return ((Record *)node)->red;
}

static CCINT32 rec_set_color(void *user, void *node, CCBOOL is_red)
{
    (void)user;
    ((Record *)node)->red = is_red;
    return 0;
}

static CCINT32 rec_left_rotate(void *user, void *node)
{
    Record *x = (Record *)node;
    Record *y = x->right;
    (void)user;
    if (!y)
        return 1;

    x->right = y->left;
    if (y->left)
        y->left->parent = x;

    y->parent = x->parent;
    if (x->parent)
    {
        if (x->parent->left == x)
            x->parent->left = y;
        else
            x->parent->right = y;
    }

    y->left = x;
    x->parent = y;
    return 0;
}

static CCINT32 rec_right_rotate(void *user, void *node)
{
    Record *x = (Record *)node;
    Record *y = x->left;
    (void)user;
    if (!y)
        return 1;

    x->left = y->right;
    if (y->right)
        y->right->parent = x;

    y->parent = x->parent;
    if (x->parent)
    {
        if (x->parent->left == x)
            x->parent->left = y;
        else
            x->parent->right = y;
    }

    y->right = x;
    x->parent = y;
    return 0;
}

static void rec_ch_root(void *user, void *new_root)
{
    ((Index *)user)->root = (Record *)new_root;
}
```

初始化和使用：

```C
static void index_init(Index *idx)
{
    memset(idx, 0, sizeof(*idx));
    idx->ops.user = idx;
    idx->ops.compare_key = rec_cmp_key;
    idx->ops.compare_node = rec_cmp_node;
    idx->ops.copy_node = rec_copy_node;
    idx->ops.find_instead = rec_find_instead;
    idx->ops.get_node = rec_get_node;
    idx->ops.set_node = rec_set_node;
    idx->ops.get_color = rec_get_color;
    idx->ops.set_color = rec_set_color;
    idx->ops.left_rotate = rec_left_rotate;
    idx->ops.right_rotate = rec_right_rotate;
    idx->ops.ch_root = rec_ch_root;
}

static Record *index_put(Index *idx, CCUINTPTR key, const char *name)
{
    Record *r = ccalloc(NULL, sizeof(*r), "index record");
    if (!r)
        return NULL;

    r->key = key;
    snprintf(r->name, sizeof(r->name), "%s", name);

    if (cc_rbt_core_insert(&idx->ops, idx->root, r) != 0)
    {
        ccalloc(r, 0, NULL);
        return NULL;
    }
    return r;
}

static void index_remove(Index *idx, CCUINTPTR key)
{
    Record *r = cc_rbt_core_remove(&idx->ops, idx->root, key);
    if (r)
        ccalloc(r, 0, NULL);
}

static CCINT32 print_record(void *data, void *user)
{
    Record *r = (Record *)data;
    (void)user;
    (void)r;
    /* 此处可输出 r->name；回调返回 0 继续遍历，非 0 中止 */
    return 0;
}

static void index_walk(Index *idx)
{
    (void)cc_rbt_core_inorder_tranverse(&idx->ops, idx->root, NULL, print_record);
}

static void free_records(Record *r)
{
    if (!r)
        return;
    free_records(r->left);
    free_records(r->right);
    ccalloc(r, 0, NULL);
}

static void index_destroy(Index *idx)
{
    free_records(idx->root);
    idx->root = NULL;
}
```

使用顺序：`index_init` → `index_put` → `index_walk` / `index_remove` → `index_destroy`。`root` 由 `ch_root` 回写，调用方直接使用 `idx.root`。

## 6. 线程

```C
#include <ccthread.h>
#include <cctime.h>
#include <string.h>

typedef struct job {
    CCLock lock;
    CCINT32 done;
    CCINT32 value;
} Job;

static void job_entry(void *data, CCThread id)
{
    Job *job = (Job *)data;
    (void)id;

    ccsleep(10000);                 /* 模拟工作 */

    cc_lock(job->lock);
    job->value = 42;
    job->done = 1;
    cc_unlock(job->lock);
}

static void run_job(void)
{
    Job job;
    CCThread t;

    memset(&job, 0, sizeof(job));
    job.lock = cc_create_lock();
    if (!job.lock)
        return;

    t = cc_thread_create(job_entry, &job);
    if (t)
        cc_thread_wait(t, 0);       /* 0：无限等待 */

    cc_lock(job.lock);
    /* 读取 job.done / job.value */
    cc_unlock(job.lock);

    cc_destroy_lock(job.lock);
}
```

线程入口收到创建时传入的 `data` 和线程自己的 ID。`cc_thread_wait` 等待目标线程结束；共享数据用 `CCLock` 保护。

## 7. 文件管理

### 7.1 挂载、读写与检索

```C
#include <ccfiles.h>

static CCINT32 print_entry(void *data, void *user)
{
    CCFsEntry *entry = (CCFsEntry *)data;
    (void)user;
    (void)entry;

    /* entry->name 只在回调期间有效；需要保留时立即拷贝 */
    /* entry->type / size / readonly / crc32 可直接读取 */
    return 0;                       /* 0 继续遍历，非 0 中断并透传 */
}

static void files_once(void)
{
    CCFiles *fs = ccfiles_create();
    char buf[32];

    if (!fs)
        return;

    if (ccfiles_mount(fs, "/assets", "./assets", CC_FS_MOUNT_DIR) == CC_FS_OK)
    {
        (void)ccfiles_create_file(fs, "/assets/notes.txt");
        (void)ccfiles_open(fs, "/assets/notes.txt");
        (void)ccfiles_write(fs, "hello", 0, 5);
        (void)ccfiles_flush(fs);

        (void)ccfiles_read(fs, buf, 0, sizeof(buf));
        (void)ccfiles_list(fs, "/assets", print_entry, NULL);
        (void)ccfiles_find(fs, "/assets", "*.txt", print_entry, NULL);
    }

    ccfiles_destroy(fs);             /* flush + close 全部打开的文件 */
}
```

`ccfiles_mount` 把真实目录挂到虚拟路径；`ccfiles_open` 之后 `ccfiles_read` / `ccfiles_write` 都带偏移量，可以随机读写。`ccfiles_destroy` 会 flush 并关闭仍然打开的文件。

### 7.2 打包与解包

```C
#include <ccfiles.h>

static void package_once(void)
{
    CCFiles *fs = ccfiles_create();
    if (!fs)
        return;

    if (ccfiles_mount(fs, "/assets", "./assets", CC_FS_MOUNT_DIR) != CC_FS_OK)
    {
        ccfiles_destroy(fs);
        return;
    }

    if (ccfiles_pack(fs, "/assets", "./assets.pkg") == CC_FS_OK)
    {
        /* 把包挂到虚拟树，再解到真实目录；包内节点只读 */
        if (ccfiles_mount(fs, "/package", "./assets.pkg", CC_FS_MOUNT_PKG) == CC_FS_OK)
            (void)ccfiles_unpack(fs, "/package", "./restored");
    }

    ccfiles_destroy(fs);
}
```

`pack` 把虚拟路径下的内容写成资源包；`unpack` 把虚拟树中的包解到真实路径。包内节点只读，适合发布资源。

## 8. 网络

网络连接默认开启 keepalive：空闲 60 秒后探测，间隔 10 秒，连续 6 次无响应判定断开。连接入口返回即释放 socket，应用通过入口内的循环决定何时结束。

### 8.1 TCP 客户端

```C
#include <ccnet.h>

static void client_entry(const CCNetTcpOps *ops, void *ctx, void *user)
{
    const char request[] = "PING\n";
    char buf[256];
    CCINT64 off = 0;

    (void)user;

    while (off < 5)                 /* 单次发送可能只写一部分 */
    {
        CCINT64 n = ops->send(ctx, request + off, (CCUINT64)(5 - off), 0);
        if (n == CC_NET_ERR_TIMEOUT)
            continue;
        if (n < 0)
            return;
        off += n;
    }

    for (;;)
    {
        CCINT64 n = ops->recv(ctx, buf, sizeof(buf), 0);
        if (n == CC_NET_ERR_TIMEOUT)
            continue;               /* 业务检查点：可在这里判断退出条件 */
        if (n <= 0)
            break;                  /* 0：对端关闭；负值：错误 */
        /* 按应用层协议处理收到的字节 */
    }
}

static void connect_example(void)
{
    CCNetAddr addrs[8];
    CCINT32 count = ccnet_dns("example.com", 9000, addrs, 8);

    for (CCINT32 i = 0; i < count; i++)
    {
        if (ccnet_tcp_connect(&addrs[i], 0, client_entry, NULL) == CC_NET_OK)
            return;                 /* 首个成功即停止，回退策略由调用方决定 */
    }
}
```

### 8.2 TCP 服务端

```C
#include <ccnet.h>

static void connection_entry(const CCNetTcpOps *ops, void *ctx, void *user)
{
    char buf[512];
    (void)user;

    for (;;)
    {
        CCINT64 n = ops->recv(ctx, buf, sizeof(buf), 0);
        if (n == CC_NET_ERR_TIMEOUT)
            continue;
        if (n <= 0)
            return;

        CCINT64 off = 0;
        while (off < n)
        {
            CCINT64 w = ops->send(ctx, buf + off, (CCUINT64)(n - off), 0);
            if (w == CC_NET_ERR_TIMEOUT)
                continue;
            if (w < 0)
                return;
            off += w;
        }
    }
}

static void accept_entry(const CCNetAcceptOps *ops, void *ctx, void *user)
{
    (void)user;
    for (;;)
    {
        CCINT32 rc = ops->accept(ctx, connection_entry, NULL, 0);
        if (rc == CC_NET_ERR_TIMEOUT)
            continue;               /* 默认 30 秒一次检查点 */
        if (rc < 0)
            return;
    }
}

static void listen_example(void)
{
    CCNetAddr addr;

    if (ccnet_addr_parse("0.0.0.0", 9000, &addr) != CC_NET_OK)
        return;

    (void)ccnet_tcp_listen(&addr, accept_entry, NULL);
}
```

`accept` 每成功一次就为连接创建一个善后线程；监听器只负责接入，连接的生命周期由 `connection_entry` 的返回决定。

### 8.3 UDP 端点

```C
#include <ccnet.h>

static void udp_entry(const CCNetUdpOps *ops, void *ctx, void *user)
{
    CCNetAddr from;
    char buf[2048];
    (void)user;

}
```

### 8.4 全双工：一个入口线程 + 一个写线程

TCP 是全双工的。入口线程负责读，另起一个协作线程负责写，两者可以同时阻塞；这比在同一个循环里“先发后收”更适合长连接。

```C
#include <ccnet.h>
#include <ccthread.h>
#include <cctime.h>
#include <stdatomic.h>

typedef struct duplex {
    const CCNetTcpOps *ops;
    void *ctx;
    _Atomic CCINT32 stop;
    _Atomic CCINT32 writer_failed;
} Duplex;

static CCBOOL sleep_or_stop(Duplex *d, CCUINT32 us)
{
    CCUINT32 elapsed = 0;
    while (elapsed < us)
    {
        if (atomic_load_explicit(&d->stop, memory_order_acquire))
            return CCFALSE;
        ccsleep(10000);
        elapsed += 10000;
    }
    return atomic_load_explicit(&d->stop, memory_order_acquire) ? CCFALSE : CCTRUE;
}

static void writer_entry(void *data, CCThread id)
{
    Duplex *d = (Duplex *)data;
    (void)id;

    while (!atomic_load_explicit(&d->stop, memory_order_acquire))
    {
        CCINT64 off = 0;
        while (off < 4)
        {
            CCINT64 n = d->ops->send(d->ctx, "tick" + off,
                                     (CCUINT64)(4 - off), 1000000);
            if (n == CC_NET_ERR_TIMEOUT)
            {
                if (atomic_load_explicit(&d->stop, memory_order_acquire))
                    return;
                continue;
            }
            if (n < 0)
            {
                /* 写方向已经不可用：发布失败并通知入口线程退出 */
                atomic_store_explicit(&d->writer_failed, 1, memory_order_release);
                atomic_store_explicit(&d->stop, 1, memory_order_release);
                return;
            }
            off += n;
        }
        if (!sleep_or_stop(d, 1000000))
            return;
    }
}

static void duplex_entry(const CCNetTcpOps *ops, void *ctx, void *user)
{
    Duplex d;
    CCThread writer;
    char buf[256];
    double last_rx = cctime();

    (void)user;

    d.ops = ops;
    d.ctx = ctx;
    atomic_init(&d.stop, 0);
    atomic_init(&d.writer_failed, 0);

    writer = cc_thread_create(writer_entry, &d);
    if (!writer)
        return;

    for (;;)
    {
        CCINT64 n;

        if (atomic_load_explicit(&d.stop, memory_order_acquire))
            break;

        /* 1 秒上限：写线程退出后，入口最多再等一次超时就能回到循环顶部检查 stop；
         * 这里不要改成 CC_NET_TIMEOUT_INFINITE。 */
        n = ops->recv(ctx, buf, sizeof(buf), 1000000);
        if (n == CC_NET_ERR_TIMEOUT)
        {
            /* 示例业务条件：30 秒没有收到任何数据就主动收尾 */
            if (cctime() - last_rx >= 30.0)
            {
                atomic_store_explicit(&d.stop, 1, memory_order_release);
                break;
            }
            continue;
        }
        if (n <= 0)
        {
            atomic_store_explicit(&d.stop, 1, memory_order_release);
            break;
        }

        last_rx = cctime();
        /* 处理对端消息；写线程同时发送，收循环专注读取 */
    }

    atomic_store_explicit(&d.stop, 1, memory_order_release);
    cc_thread_wait(writer, 0);      /* 必须在入口返回前结束协作线程 */

    if (atomic_load_explicit(&d.writer_failed, memory_order_acquire))
    {
        /* 写线程失败：记录日志、重连或让入口返回，由业务决定 */
    }
}
```

要点：

- 同一时刻一个读方向、一个写方向可以并发；同一方向仍由调用方串行化。
- 写线程只使用 `ctx` 的写操作，入口线程只使用读操作，`ctx` 在入口返回前一直有效。
- 入口返回前必须 `cc_thread_wait` 等待写线程退出；协作线程由调用方回收。
- 入口的 `recv` 使用 1 秒有限超时：写线程退出后，入口最多再等一次接收超时就能检查 `stop`；写线程的发送超时同样是 1 秒，因此 `cc_thread_wait` 的等待有上限。示例不能改成 `CC_NET_TIMEOUT_INFINITE`。
- 写线程在发送超时分支检查 `stop`，在发送错误时发布 `writer_failed` 并置 `stop`；入口线程据此退出并等待写线程。
- 示例用“30 秒没有收到数据”作为业务停止条件；接入真实协议时替换成自己的退出判断。
- 连接空闲时的半开检测由默认 keepalive 兜底；需要更快检测业务失活时，在 `recv` 的 `TIMEOUT` 分支里加应用层心跳判断。
